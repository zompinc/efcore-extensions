SELECT "t"."Id", CAST(COALESCE(SUM(CAST("t"."Id" AS INTEGER)) OVER(PARTITION BY "t"."Id" / 10 ), 0) AS INTEGER) AS "Total", CAST(COALESCE(SUM(CAST("t"."Col1" AS INTEGER)) OVER(PARTITION BY "t"."Id" / 10 ), 0) AS INTEGER) AS "TotalCol1", CAST(MIN("t"."Col1") OVER(PARTITION BY "t"."Id" / 10 ) AS INTEGER) AS "Min", CAST(MAX("t"."Id") OVER(PARTITION BY "t"."Id" / 10 ) AS INTEGER) AS "Max"
FROM "TestRows" AS "t"
ORDER BY "t"."Id"
