SELECT "t2"."P0"
FROM (
    SELECT ROW_NUMBER() OVER(ORDER BY "t1"."P0") AS "P0"
    FROM (
        SELECT CAST(MAX("t0"."P0") OVER() AS INTEGER) AS "P0"
        FROM (
            SELECT CAST(MAX("t"."Id") OVER() AS INTEGER) AS "P0"
            FROM "TestRows" AS "t"
        ) AS "t0"
    ) AS "t1"
) AS "t2"
