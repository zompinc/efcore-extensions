SELECT "t0"."Id"
FROM (
    SELECT "t"."Id", CAST(max(AVG(CAST("t"."Col1" AS REAL) * CAST("t"."Col1" AS REAL)) OVER(PARTITION BY "t"."Id" / 10 ) - AVG(CAST("t"."Col1" AS REAL)) OVER(PARTITION BY "t"."Id" / 10 ) * AVG(CAST("t"."Col1" AS REAL)) OVER(PARTITION BY "t"."Id" / 10 ), 0.0) AS REAL) AS "P0"
    FROM "TestRows" AS "t"
) AS "t0"
WHERE "t0"."P0" > 0.0
ORDER BY "t0"."Id"
