SELECT "s"."Id", "s"."RowNumber"
FROM (
    SELECT "t1"."Id", "t1"."RowNumber"
    FROM "TestRows" AS "t"
    INNER JOIN (
        SELECT "t0"."Id", ROW_NUMBER() OVER(PARTITION BY "t0"."Col1" ORDER BY "t0"."Date") AS "RowNumber"
        FROM "TestRows" AS "t0"
    ) AS "t1" ON "t"."Id" = "t1"."Id"
) AS "s"
WHERE "s"."Id" <> -999
