SELECT [t].[Id], CAST(COALESCE(SUM(CAST([t].[Id] AS bigint)) OVER(PARTITION BY [t].[Id] / 10 ), 0) AS int) AS [Total], MIN([t].[Col1]) OVER(PARTITION BY [t].[Id] / 10 ) AS [Min], MAX([t].[Id]) OVER(PARTITION BY [t].[Id] / 10 ) AS [Max], AVG(CAST([t].[Id] AS float)) OVER(PARTITION BY [t].[Id] / 10 ) AS [Average]
FROM [TestRows] AS [t]
ORDER BY [t].[Id]
