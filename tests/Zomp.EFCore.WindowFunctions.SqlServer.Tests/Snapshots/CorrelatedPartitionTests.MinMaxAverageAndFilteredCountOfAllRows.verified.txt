SELECT [t].[Id], (
    SELECT COUNT(*)
    FROM [TestRows] AS [t0]
    WHERE [t0].[Id] > 2) AS [Total], (
    SELECT MIN([t1].[Col1])
    FROM [TestRows] AS [t1]
    WHERE [t1].[Id] > 2) AS [Min], (
    SELECT MAX([t2].[Id])
    FROM [TestRows] AS [t2]
    WHERE [t2].[Id] > 2) AS [Max], (
    SELECT AVG(CAST([t3].[Id] AS float))
    FROM [TestRows] AS [t3]
    WHERE [t3].[Id] > 2) AS [Average]
FROM [TestRows] AS [t]
WHERE [t].[Id] > 2
ORDER BY [t].[Id]
