SELECT [t0].[Id], [t0].[Col1]
FROM (
    SELECT [t].[Id], [t].[Col1], SUM(CASE
        WHEN [t].[Id] > 12 THEN 0
        ELSE 1
    END) OVER(ORDER BY [t].[Id] DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS [Index]
    FROM [TestRows] AS [t]
) AS [t0]
WHERE [t0].[Index] = 0
ORDER BY [t0].[Id] DESC
