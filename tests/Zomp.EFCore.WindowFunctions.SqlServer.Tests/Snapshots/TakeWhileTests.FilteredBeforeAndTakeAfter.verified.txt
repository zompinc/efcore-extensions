DECLARE @p int = 2;

SELECT TOP(@p) [t0].[Id]
FROM (
    SELECT [t].[Id], SUM(CASE
        WHEN [t].[Id] < 7 THEN 0
        ELSE 1
    END) OVER(ORDER BY [t].[Id] ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS [Index]
    FROM [TestRows] AS [t]
    WHERE [t].[Id] > 2
) AS [t0]
WHERE [t0].[Index] > 0
ORDER BY [t0].[Id]
