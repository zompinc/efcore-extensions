SELECT [t1].[Id], [t1].[Col1], [t1].[Date], [t1].[IdBytes], [t1].[SomeGuid]
FROM (
    SELECT [t0].[Id], [t0].[Col1], [t0].[Date], [t0].[IdBytes], [t0].[SomeGuid], ROW_NUMBER() OVER(ORDER BY [t0].[P0]) AS [P0]
    FROM (
        SELECT [t].[Id], [t].[Col1], [t].[Date], [t].[IdBytes], [t].[SomeGuid], MAX([t].[Id]) OVER() AS [P0]
        FROM [TestRows] AS [t]
    ) AS [t0]
) AS [t1]
ORDER BY [t1].[P0]
