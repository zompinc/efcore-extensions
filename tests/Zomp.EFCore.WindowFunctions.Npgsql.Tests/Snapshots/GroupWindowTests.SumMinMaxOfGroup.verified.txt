SELECT t."Id", COALESCE(SUM(t."Id"::bigint) OVER(PARTITION BY t."Id" / 10 ), 0)::int AS "Total", COALESCE(SUM(t."Col1"::bigint) OVER(PARTITION BY t."Id" / 10 ), 0)::int AS "TotalCol1", MIN(t."Col1") OVER(PARTITION BY t."Id" / 10 ) AS "Min", MAX(t."Id") OVER(PARTITION BY t."Id" / 10 ) AS "Max"
FROM "TestRows" AS t
ORDER BY t."Id"
