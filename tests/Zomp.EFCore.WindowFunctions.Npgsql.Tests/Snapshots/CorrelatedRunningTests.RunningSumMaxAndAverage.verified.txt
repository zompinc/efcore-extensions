SELECT t."Id", (
    SELECT COALESCE(sum(t0."Id"), 0)::int
    FROM "TestRows" AS t0
    WHERE t0."Id" <= t."Id") AS "RunningTotal", (
    SELECT max(t1."Col1")
    FROM "TestRows" AS t1
    WHERE t1."Id" <= t."Id") AS "RunningMax", (
    SELECT avg(t2."Id"::double precision)
    FROM "TestRows" AS t2
    WHERE t2."Id" <= t."Id") AS "RunningAverage"
FROM "TestRows" AS t
ORDER BY t."Id"
