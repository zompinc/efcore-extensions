SELECT t0."Id", t0."Index", t0."InGroup"
FROM (
    SELECT t."Id", CAST(ROW_NUMBER() OVER(PARTITION BY t."Id" / 10 ORDER BY t."Id") - 1 AS integer) AS "Index", CAST(COUNT(*) OVER(PARTITION BY t."Id" / 10 ) AS integer) AS "InGroup"
    FROM "TestRows" AS t
    WHERE t."Id" > 2
) AS t0
WHERE t0."InGroup" > 1
ORDER BY t0."Id"
